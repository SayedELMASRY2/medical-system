Pydantic Shared State Schema (`AgentState`)

Parallel-Safe Reducers 
from typing import Annotated, List, Optional, Dict, Any
from pydantic import BaseModel, Field
import operator

class AgentState(BaseModel):
    # 1. Patient Context
    patient_id: str
    gestational_age_weeks: int
    current_vitals: VitalSigns
    reported_symptoms: List[str]
    longitudinal_history: List[Dict[str, Any]]

    # 2. Pipeline Results
    timeline_trends: Optional[List[TrendFeature]] = None
    deterministic_risk_level: Optional[str] = None  # "LOW", "MONITOR", "CLINICAL_REVIEW", "HIGH_PRIORITY"
    evaluated_rules: Optional[List[RuleResult]] = None
    retrieved_citations: Optional[List[Citation]] = None
    triage_explanation: Optional[str] = None

    # 3. Parallel Accumulators (Safe from race conditions)
    agent_outputs: Annotated[List[AgentOutput], operator.add] = Field(default_factory=list)
    agent_errors: Annotated[List[str], operator.add] = Field(default_factory=list)

    # 4. Lifecycle Execution Meta
    execution_id: str
    status: str = "IN_PROGRESS"  # "IN_PROGRESS", "SUCCESS", "DEGRADED", "FAILED"

LangGraph Python Graph Blueprint

Typed StateGraph 
from langgraph.graph import StateGraph, END

# Initialize Graph
workflow = StateGraph(AgentState)

# Add Nodes
workflow.add_node("supervisor", supervisor_node)
workflow.add_node("timeline_agent", timeline_agent_node)
workflow.add_node("risk_engine", deterministic_risk_engine_node)
workflow.add_node("triage_agent", triage_agent_node)
workflow.add_node("rag_agent", rag_agent_node)
workflow.add_node("synthesizer", explanation_synthesizer_node)

# Flow Edges
workflow.set_entry_point("supervisor")
workflow.add_edge("supervisor", "timeline_agent")
workflow.add_edge("timeline_agent", "risk_engine")

# Conditional Branching
def route_after_risk(state: AgentState):
    if state.deterministic_risk_level in ["MONITOR", "CLINICAL_REVIEW", "HIGH_PRIORITY"]:
        return ["triage_agent", "rag_agent"]  # Parallel Fan-Out
    return ["synthesizer"]

workflow.add_conditional_edges("risk_engine", route_after_risk, {
    "triage_agent": "triage_agent",
    "rag_agent": "rag_agent",
    "synthesizer": "synthesizer"
})

# Fan-in Edges
workflow.add_edge("triage_agent", "synthesizer")
workflow.add_edge("rag_agent", "synthesizer")
workflow.add_edge("synthesizer", END)

app = workflow.compile()

CDSS Safety Guardrails & Graceful Degradation

Fail-Safe Matrix
Failure Scenario Subsystem Impact Graceful Degradation Behavior
LLM API Outage (OpenAI Down) Triage / Timeline agents fail to generate natural language. System bypasses LLM entirely and presents the deterministic risk level directly with a banner: "AI explanation unavailable — deterministic rule triggered: BP > 140/90."
Vector DB / RAG Timeout pgvector search does not respond within 3000ms. RAG node returns empty citations; Triage Agent synthesizes explanation based on standard clinical knowledge without specific literature citations.
Hallucination / Override Attempt Prompt injection attempts to downgrade 🔴 HIGH PRIORITY to 🟢 LOW. Immutable Guardrail: Output schemas forbid LLMs from altering deterministic_risk_level. The risk flag in the database is strictly write-locked to the Rule Engine.
Prescription Attempt LLM recommends specific medications or drug dosages. Post-processing filter scrubs drug recommendations and replaces them with standard CDSS prompt: "Refer to attending obstetrician for pharmaceutical management."
